country_event = {
	id = GER_prewar.the_die_is_cast
	title = GER_prewar.the_die_is_cast.t
	desc = GER_prewar.the_die_is_cast.d
	picture = GFX_news_event_001
	is_triggered_only = yes
	option = {
		name = GER_prewar.the_die_is_cast.a
		add_war_support = 0.03
		set_country_flag = GER_war_declared_on_soviets
		if = {
			limit = {
				SOV = { exists = yes }
				NOT = { has_war_with = SOV }
			}
			SOV = {
				remove_from_faction = FRA
				remove_from_faction = ITA
			}
			if = {
				limit = { FRA = { exists = yes } }
				FRA = {
					remove_from_faction = SOV
					set_country_flag = FRA_hesitating_on_war
					country_event = { id = FRA_german_war.1 days = 2 }
					country_event = { id = FRA_german_war.decision days = 60 }
				}
			}
			if = {
				limit = { ITA = { exists = yes } }
				ITA = {
					remove_from_faction = SOV
					set_country_flag = ITA_hesitating_on_war
					country_event = { id = ITA_german_war.1 days = 2 }
					country_event = { id = ITA_german_war.decision days = 100 }
				}
			}
			declare_war_on = {
				target = SOV
				type = annex_everything
			}
			SOV = {
				country_event = {
					id = SOV_german_war.1
					days = 1
				}
			}
		}
	}
}
