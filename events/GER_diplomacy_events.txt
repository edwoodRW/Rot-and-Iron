	picture = GFX_news_event_001
	is_triggered_only = yes
	option = {
		name = FRA_german_war.decision.a
		ai_chance = {
			base = 3
			modifier = {
				factor = 0.1
				GER = { has_country_flag = GER_bordeaux_signed }
			}
		}
		clr_country_flag = FRA_hesitating_on_war
		SOV = { add_to_faction = FRA }
		if = {
			limit = { NOT = { has_war_with = GER } }
			declare_war_on = {
				target = GER
				type = take_territory
			}
		}
		add_war_support = 0.05
	}
	option = {
		name = FRA_german_war.decision.b
		ai_chance = {
			base = 1
			modifier = {
				factor = 5

			}
		}
		clr_country_flag = FRA_hesitating_on_war
		set_country_flag = FRA_stayed_out_of_war
		add_stability = 0.03
	}
}

	title = SOV_german_ceasefire.1.t
	desc = SOV_german_ceasefire.1.d
	picture = GFX_news_event_001
	is_triggered_only = yes
	option = {
		name = SOV_german_ceasefire.1.a
		trigger = {
			has_war_with = GER
			GER = {
				exists = yes
				has_country_flag = GER_ceasefire_talks_active
				has_country_flag = RI_GER_ceasefire_pending
				NOT = { has_country_flag = RI_GER_war_crisis }
			}
		}
		add_stability = 0.03
		add_war_support = -0.03
		set_country_flag = SOV_accepts_ceasefire_talks
		GER = {
			clr_country_flag = RI_GER_ceasefire_pending
			clr_country_flag = RI_GER_ceasefire_refused
		}
	}
	option = {
		name = SOV_german_ceasefire.1.b
		trigger = {
			has_war_with = GER
			GER = {
				exists = yes
				has_country_flag = GER_ceasefire_talks_active
				has_country_flag = RI_GER_ceasefire_pending
				NOT = { has_country_flag = RI_GER_war_crisis }
			}
		}
		add_war_support = 0.03
		add_stability = -0.02
		clr_country_flag = SOV_accepts_ceasefire_talks
		GER = {
			clr_country_flag = RI_GER_ceasefire_pending
			set_country_flag = RI_GER_ceasefire_refused
		}
	}
	option = {
		name = SOV_german_ceasefire.1.c
		trigger = {
			OR = {
				NOT = { has_war_with = GER }
				NOT = { GER = { exists = yes } }
				GER = { has_country_flag = RI_GER_war_crisis }
				GER = { NOT = { has_country_flag = GER_ceasefire_talks_active } }
				GER = { NOT = { has_country_flag = RI_GER_ceasefire_pending } }
			}
		}
	}
