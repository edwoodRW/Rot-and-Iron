###########################################################
# ELECTION RESULT - hidden dispatcher
# Weights are placeholders keyed off flags you already set
###########################################################

country_event = {
	id = CAN_election.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		set_country_flag = CAN_election_held

		random_list = {
			30 = {
				modifier = {
					add = 20
					has_country_flag = CAN_detroit_agreement_signed
				}
				country_event = { id = CAN_carney.1 }
			}
			30 = {
				modifier = {
					add = 15
					has_country_flag = CAN_black_angus_understanding
				}
				modifier = {
					add = 10
					has_country_flag = CAN_conscription_repealed
				}
				country_event = { id = CAN_angus.1 }
			}
			40 = {
				modifier = {
					add = 20
					has_country_flag = CAN_boris_national_figure
				}
				modifier = {
					add = 10
					has_country_flag = CAN_boris_deputy_pm_appointed
				}
				country_event = { id = CAN_borispm.1 }
			}
		}
	}
}
