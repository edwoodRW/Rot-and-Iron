		GER = {
			clr_country_flag = RI_GER_urals_offer_pending
			country_event = { id = RI_GER_eastern_campaign.4 hours = 1 }

	}
	option = {
		name = RI_GER_eastern_campaign.3.c

			OR = {

			}
		}
		GER = { clr_country_flag = RI_GER_urals_offer_pending }
