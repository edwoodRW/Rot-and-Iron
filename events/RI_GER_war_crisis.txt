country_event = {
	id = RI_GER_crisis.8
	title = RI_GER_crisis.8.t
	desc = RI_GER_crisis.8.d
	picture = GFX_report_event_Quiet_Words_in_Brussels
	is_triggered_only = yes

	immediate = {
		random_country = {
			limit = { RI_SOV_german_ally_settlement_ready = yes }
			save_event_target_as = RI_SOV_defeated_ally
		}
	}

	option = {
		name = RI_GER_crisis.8.puppet
		trigger = {
			country_exists = event_target:RI_SOV_defeated_ally
			event_target:RI_SOV_defeated_ally = { RI_SOV_german_ally_settlement_ready = yes }
		}
		RI_SOV_end_german_ally_war = yes
		if = {
			limit = {
				country_exists = event_target:RI_SOV_defeated_ally
				event_target:RI_SOV_defeated_ally = { has_war = no NOT = { is_in_peace_conference = yes } }
			}
			event_target:RI_SOV_defeated_ally = {
				if = { limit = { is_in_faction = yes } leave_faction = yes }
				if = { limit = { is_subject_of = GER } GER = { end_puppet = PREV } }
				clr_country_flag = RI_SOV_german_war_ally
				set_politics = { ruling_party = communist elections_allowed = no }
				if = { limit = { tag = POL } set_cosmetic_tag = RI_POL_soviet }
			}
			puppet = { target = event_target:RI_SOV_defeated_ally }
			if = {
				limit = { event_target:RI_SOV_defeated_ally = { tag = POL } }
				GER = {
					RI_GER_return_ermland_to_soviet = yes
					if = {
						limit = { RI_GER_soviet_settlement_cleanup_ready = yes }
						RI_GER_finalize_soviet_settlement = yes
					}
				}
			}
		}
		else = { country_event = { id = RI_GER_crisis.9 hours = 1 } }
		clr_country_flag = RI_SOV_german_ally_event_pending
		ai_chance = { factor = 80 }
	}
	option = {
		name = RI_GER_crisis.8.annex
		trigger = {
			country_exists = event_target:RI_SOV_defeated_ally
			event_target:RI_SOV_defeated_ally = {
				RI_SOV_german_ally_settlement_ready = yes
				NOT = { any_subject_country = { exists = yes } }
			}
		}
		RI_SOV_end_german_ally_war = yes
		if = {
			limit = {
				country_exists = event_target:RI_SOV_defeated_ally
				event_target:RI_SOV_defeated_ally = { has_war = no NOT = { is_in_peace_conference = yes } }
			}
			event_target:RI_SOV_defeated_ally = {
				if = { limit = { is_in_faction = yes } leave_faction = yes }
				if = { limit = { is_subject_of = GER } GER = { end_puppet = PREV } }
				clr_country_flag = RI_SOV_german_war_ally
			}
			annex_country = { target = event_target:RI_SOV_defeated_ally transfer_troops = no }
		}
		else = { country_event = { id = RI_GER_crisis.9 hours = 1 } }
		clr_country_flag = RI_SOV_german_ally_event_pending
		ai_chance = { factor = 20 }
	}
	option = {
		name = RI_GER_crisis.8.unavailable
		trigger = {
			NOT = {
				country_exists = event_target:RI_SOV_defeated_ally
				event_target:RI_SOV_defeated_ally = { RI_SOV_german_ally_settlement_ready = yes }
			}
		}
		clr_country_flag = RI_SOV_german_ally_event_pending
	}
}
