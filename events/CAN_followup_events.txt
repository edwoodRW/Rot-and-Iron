country_event = {
    id = CAN_carney_aftermath.1
    title = CAN_carney_aftermath.1.t
    desc = CAN_carney_aftermath.1.d
    picture = GFX_report_event_carney_office
    is_triggered_only = yes

    option = {
        name = CAN_carney_aftermath.1.a
        
        # Economic bonuses
		add_ideas = CAN_carney_government
		add_ideas = CAN_detroit_framework
        
        # Political changes
        add_political_power = 150
        add_stability = 0.10
        add_war_support = -0.05
        
        # Economic modifiers
		set_country_flag = CAN_technocratic_government
        
        # Relations with USA improve
        USA = {
            add_opinion_modifier = {
                target = CAN
                modifier = CAN_carney_administration
            }
        }
        
        # Trigger follow-up economic reform events
        country_event = {
            id = CAN_carney_reforms.1
            days = 7
        }
    }
}
