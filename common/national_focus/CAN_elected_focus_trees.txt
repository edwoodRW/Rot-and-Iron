### common/national_focus/CAN_elected_focus_trees.txt
###
### Three trees that are NEVER assigned at game start (factor = 0).
### They are swapped in by load_focus_tree in the CAN_carney.1 / CAN_angus.1 /
### CAN_borispm.1 events once the election result is known.
###
### Each tree is seeded with the focuses your existing follow-up events already
### expect (the .2 / .3 events say "fires when the player completes ...").
### Replace the placeholder rewards, icons and layout as you build them out.
###
### Localisation needed for every focus id: <id> and <id>_desc

########################################################################
# CARNEY - economic reform / Detroit framework government
########################################################################
focus_tree = {
	id = canadian_carney

	country = {
		factor = 0
	}

	default = no

	focus = {
		id = CAN_carney_implement_reforms
		icon = GFX_goal_unknown
		x = 2
		y = 0
		cost = 5

		completion_reward = {
			# CAN_carney_economic_reforms.1 grants the idea, so guard against
			# double-firing if the Carney negotiation path already ran it
			if = {
				limit = { NOT = { has_idea = CAN_economic_optimization } }
				country_event = { id = CAN_carney_economic_reforms.1 }
			}
		}
		ai_will_do = { factor = 10 }
	}

	focus = {
		id = CAN_carney_deregulate_the_markets
		icon = GFX_goal_unknown
		prerequisite = { focus = CAN_carney_implement_reforms }
		x = 0
		y = 1
		cost = 5

		completion_reward = {
			country_event = { id = CAN_carney_economic_reforms.2 }
		}
		ai_will_do = { factor = 10 }
	}

	focus = {
		id = CAN_carney_attract_foreign_investment
		icon = GFX_goal_unknown
		prerequisite = { focus = CAN_carney_implement_reforms }
		x = 4
		y = 1
		cost = 5

		completion_reward = {
			country_event = { id = CAN_carney_economic_reforms.3 }
		}
		ai_will_do = { factor = 10 }
	}
}
