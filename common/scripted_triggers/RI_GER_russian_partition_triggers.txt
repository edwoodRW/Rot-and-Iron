RI_GER_partition_ready = {
	NOT = { has_country_flag = RI_GER_russian_partition_complete }
	NOT = { is_subject = yes }
	NOT = { has_capitulated = yes }
	has_war = no
	NOT = { is_in_peace_conference = yes }
	OR = {
		SOV = { exists = no }
		SOV = {
			has_war = no
			is_subject = no
			NOT = { is_in_peace_conference = yes }
			NOT = {
				any_owned_state = {
					OR = {
						NOT = { RI_GER_partition_region_state = yes }
						NOT = { is_fully_controlled_by = SOV }
						is_core_of = GER
					}
				}
			}
		}
	}
	NOT = {
		any_owned_state = {
			RI_GER_partition_region_state = yes
			is_core_of = SOV
			NOT = { is_core_of = GER }
			NOT = { is_fully_controlled_by = GER }
		}
	}
	RWE = { exists = no }
	RVN = { exists = no }
	RDN = { exists = no }
	RCA = { exists = no }
	RVG = { exists = no }
	RSW = { exists = no }
	RTS = { exists = no }
	RAI = { exists = no }
	RIR = { exists = no }
	RBU = { exists = no }
	RYK = { exists = no }
	RFA = { exists = no }
	any_state = {
		RI_GER_partition_region_state = yes
		RI_GER_partition_transferable_state = yes
	}
}
