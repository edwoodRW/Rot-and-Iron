###########################################################
# HOW THESE WORK IN HOI4
#
# HOI4 does not support true variables the way a programming
# language does. Instead we use a combination of:
#
# 1. set_variable / add_to_variable — for numeric tracking
# 2. scripted_effects — for reusable effect blocks
# 3. scripted_triggers — for reusable condition checks
# 4. country_flags — for binary on/off states
#
# The variables below are initialised in the country
# history file and modified by events and focuses.
# Threshold checks fire events when values cross key points.
###########################################################

###########################################################
# VARIABLE INITIALISATION
# Add this block to history/countries/CAN - Canada.txt
# so variables exist from game start
#
# set_variable = { CAN_mogg_faction_influence = 60 }
# set_variable = { CAN_boris_loyalty = 50 }
# set_variable = { CAN_black_authority = 70 }
# set_variable = { CAN_coalition_stability = 65 }
###########################################################

###########################################################
# MOGG FACTION INFLUENCE
# Tracks how much political influence Rees-Mogg's
# Imperial Restoration Party holds within the coalition
# Range: 0 to 100
# Starting value: 60
# 80+: Mogg Plan focuses unlock, Mogg dominant flag set
# 40-: Black Plan focuses unlock faster
# 20-: IRP departure warning fires
###########################################################

CAN_mogg_faction_influence_add = {
	add_to_variable = {
		var = CAN_mogg_faction_influence
		value = CAN_mogg_faction_influence_change
	}
	clamp_variable = {
		var = CAN_mogg_faction_influence
		min = 0
		max = 100
	}
	if = {
		limit = {
			check_variable = { CAN_mogg_faction_influence < 20 }
			NOT = { has_country_flag = CAN_coalition_broken }
			NOT = { has_country_flag = CAN_mogg_resigned }
			NOT = { has_country_flag = CAN_irp_departure_warned }
		}
		set_country_flag = CAN_irp_departure_warned
		country_event = {
			id = CAN_coalition.irp_warning
			days = 7
		}
	}
	if = {
		limit = {
			check_variable = { CAN_mogg_faction_influence > 80 }
			NOT = { has_country_flag = CAN_mogg_dominant }
		}
		set_country_flag = CAN_mogg_dominant
		country_event = {
			id = CAN_coalition.mogg_dominant
			days = 7
		}
	}
}

###########################################################
# COALITION STABILITY
# Tracks overall health of the governing coalition
# Range: 0 to 100
# Starting value: 65
# 80+: Coalition stable, no defection risk
# 40-: Defection events can fire
# 20-: Coalition collapse imminent
# 0: Coalition collapses, CAN_sovereignty.6 fires
###########################################################

CAN_coalition_stability_add = {
	add_to_variable = {
		var = CAN_coalition_stability
		value = CAN_coalition_stability_change
	}
	clamp_variable = {
		var = CAN_coalition_stability
		min = 0
		max = 100
	}
	if = {
		limit = {
			check_variable = { CAN_coalition_stability < 40 }
			NOT = { has_country_flag = CAN_coalition_wobbling }
			NOT = { has_country_flag = CAN_coalition_broken }
		}
		set_country_flag = CAN_coalition_wobbling
		country_event = {
			id = CAN_coalition.wobbling
			days = 7
		}
	}
	if = {
		limit = {
			check_variable = { CAN_coalition_stability < 1 }
			NOT = { has_country_flag = CAN_coalition_broken }
		}
		set_country_flag = CAN_coalition_broken
		country_event = {
			id = CAN_sovereignty.6
			days = 3
		}
	}
}
