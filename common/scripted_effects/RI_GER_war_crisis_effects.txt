RI_GER_check_war_crisis = {
	if = {
		limit = {
			exists = yes
			has_capitulated = no
			has_war_with = SOV
			NOT = { surrender_progress < 0.40 }
			NOT = { has_country_flag = RI_GER_war_crisis }
			any_owned_state = {
				RI_GER_state_occupied_by_soviet_coalition = yes
			}
		}
		set_country_flag = RI_GER_war_crisis
		RI_GER_wagenknecht_takes_power = yes
		country_event = { id = RI_GER_crisis.1 hours = 1 }
	}
	if = {
		limit = {
			has_war_with = SOV
			SOV = { exists = yes }
			NOT = { has_country_flag = RI_GER_soviet_takeover }
			64 = { is_owned_by = GER }
		}
		set_variable = { RI_GER_core_state_count = 0 }
		set_variable = { RI_GER_occupied_core_count = 0 }
		every_owned_state = {
			limit = { is_core_of = GER }
			add_to_variable = { PREV.RI_GER_core_state_count = 1 }
			if = {
				limit = { RI_GER_state_occupied_by_soviet_coalition = yes }
				add_to_variable = { PREV.RI_GER_occupied_core_count = 1 }
			}
		}
		multiply_variable = { RI_GER_core_state_count = 4 }
		multiply_variable = { RI_GER_occupied_core_count = 5 }
		if = {
			limit = {
				check_variable = { RI_GER_core_state_count > 0 }
				check_variable = { var = RI_GER_occupied_core_count value = RI_GER_core_state_count compare = greater_than_or_equals }
			}
			RI_GER_soviet_takeover = yes
		}
		clear_variable = RI_GER_core_state_count
		clear_variable = RI_GER_occupied_core_count
	}
}

RI_GER_soviet_takeover = {
	every_country = {
		limit = {
			NOT = { tag = GER }
			has_war_together_with = GER
			has_war_with = SOV
			capital_scope = { is_on_continent = europe }
		}
		set_country_flag = RI_SOV_german_war_ally
	}
	set_country_flag = RI_GER_soviet_takeover
	clr_country_flag = RI_GER_victory_pending
	clr_country_flag = RI_GER_recovery_unlocked
	set_country_flag = GER_liebich_falls
	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
	}
	every_owned_state = {
		limit = {
			NOT = { state = 64 }
			NOT = { RI_GER_partition_north_state = yes }
			NOT = { RI_GER_partition_west_state = yes }
			NOT = { RI_GER_partition_south_state = yes }
			NOT = { RI_GER_partition_austrian_state = yes }
			NOT = { RI_GER_partition_neighbor_state = yes }
			OR = {
				RI_GER_state_occupied_by_soviet_coalition = yes
				AND = {
					RI_GER_partition_east_state = yes
					is_fully_controlled_by = GER
				}
			}
		}
		transfer_state_to = SOV
	}
	every_country = {
		limit = {
			has_war_with = GER
			OR = { tag = SOV has_war_together_with = SOV }
		}
		set_country_flag = RI_GER_takeover_soviet_side
	}
	every_country = {
		limit = { has_country_flag = RI_GER_takeover_soviet_side }
		if = {
			limit = { has_war_with = GER }
			white_peace = { tag = GER }
		}
		clr_country_flag = RI_GER_takeover_soviet_side
	}
	SOV = { puppet = GER }
	set_capital = { state = 64 }
	set_cosmetic_tag = RI_GER_north_german
	RI_GER_establish_oder_republic = yes
	if = {
		limit = { has_country_flag = RI_GER_soviet_takeover_test RI_GER_test_polish_ssr_ready = yes }
		RI_GER_test_polish_ssr = yes
	}
	clr_country_flag = RI_GER_soviet_takeover_test
	RI_GER_return_ermland_to_soviet = yes
	if = {
		limit = { RI_GER_soviet_settlement_cleanup_ready = yes }
		RI_GER_finalize_soviet_settlement = yes
	}
	create_country_leader = {
		name = "Werner Dahl"
		desc = RI_GER_werner_dahl_desc
		picture = "gfx/leaders/leader_unknown.dds"
		ideology = communist
		traits = { }
	}
	set_politics = {
		ruling_party = communist
		elections_allowed = no
	}
	load_focus_tree = { tree = GER_soviet_occupation_focus keep_completed = no }
	country_event = { id = RI_GER_crisis.3 hours = 1 }
}

RI_SOV_check_german_war_allies = {
	every_country = {
		limit = { has_country_flag = RI_SOV_german_war_ally }
		if = {
			limit = { NOT = { has_war_with = SOV } }
			clr_country_flag = RI_SOV_german_war_ally
		}
		else = {
			set_variable = { RI_SOV_ally_core_count = 0 }
			set_variable = { RI_SOV_ally_occupied_count = 0 }
			every_owned_state = {
				limit = { is_core_of = PREV }
				add_to_variable = { PREV.RI_SOV_ally_core_count = 1 }
				if = {
					limit = { is_fully_controlled_by = SOV }
					add_to_variable = { PREV.RI_SOV_ally_occupied_count = 1 }
				}
			}
			multiply_variable = { RI_SOV_ally_core_count = 4 }
			multiply_variable = { RI_SOV_ally_occupied_count = 5 }
		}
	}
	if = {
		limit = {
			NOT = { has_country_flag = RI_SOV_german_ally_event_pending }
			any_country = { RI_SOV_german_ally_settlement_ready = yes }
		}
		set_country_flag = RI_SOV_german_ally_event_pending
		country_event = { id = RI_GER_crisis.8 hours = 1 }
	}
}

RI_GER_end_soviet_coalition_war = {
	# Record both sides before individual peace deals change their war relationships.
	every_country = {
		limit = {
			OR = {
				tag = GER
				AND = {
					has_war_together_with = GER
					has_war_with = SOV
				}
			}
		}
		set_country_flag = RI_GER_settlement_german_side
	}
	every_country = {
		limit = {
			OR = {
				tag = SOV
				AND = {
					has_war_together_with = SOV
					has_war_with = GER
				}
			}
		}
		set_country_flag = RI_GER_settlement_soviet_side
	}
	every_country = {
		limit = { has_country_flag = RI_GER_settlement_german_side }
		every_country = {
			limit = {
				has_country_flag = RI_GER_settlement_soviet_side
				has_war_with = PREV
			}
			white_peace = { tag = PREV }
		}
	}
	every_country = {
		limit = {
			OR = {
				has_country_flag = RI_GER_settlement_german_side
				has_country_flag = RI_GER_settlement_soviet_side
			}
		}
		clr_country_flag = RI_GER_settlement_german_side
		clr_country_flag = RI_GER_settlement_soviet_side
	}
	if = {
		limit = { GER = { NOT = { has_war_with = SOV } } }
		GER = { set_country_flag = RI_GER_settlement_peace_completed }
	}
}
